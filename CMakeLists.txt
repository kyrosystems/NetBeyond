cmake_minimum_required(VERSION 3.16)
project(NetBeyond C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(nb_core src/core.c src/history.c)
target_include_directories(nb_core PUBLIC src)
if(WIN32)
  add_executable(NetBeyond WIN32 src/netbeyond.c)
  target_link_libraries(NetBeyond PRIVATE nb_core ole32 oleaut32 uuid)
endif()
add_executable(test_core tests/test_core.c)
target_link_libraries(test_core PRIVATE nb_core)
add_executable(test_history tests/test_history.c)
target_link_libraries(test_history PRIVATE nb_core)
enable_testing()
add_test(NAME core COMMAND test_core)
add_test(NAME history COMMAND test_history)
