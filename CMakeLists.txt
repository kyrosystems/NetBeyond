cmake_minimum_required(VERSION 3.16)
project(NetBeyond C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(nb_core src/core.c src/history.c src/html.c src/layout.c src/http.c)
target_include_directories(nb_core PUBLIC src)
if(WIN32)
  target_sources(nb_core PRIVATE src/network.c)
  add_executable(NetBeyond WIN32 src/netbeyond.c)
  target_link_libraries(NetBeyond PRIVATE nb_core winhttp)
endif()
foreach(test_name core history html layout http)
  add_executable(test_${test_name} tests/test_${test_name}.c)
  target_link_libraries(test_${test_name} PRIVATE nb_core)
endforeach()
enable_testing()
foreach(test_name core history html layout http)
  add_test(NAME ${test_name} COMMAND test_${test_name})
endforeach()
