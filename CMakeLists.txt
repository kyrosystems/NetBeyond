cmake_minimum_required(VERSION 3.16)
project(NetBeyond C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
include(FetchContent)
if(WIN32)
  set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
  set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
  set(USE_SHARED_MBEDTLS_LIBRARY OFF CACHE BOOL "" FORCE)
  set(USE_STATIC_MBEDTLS_LIBRARY ON CACHE BOOL "" FORCE)
  FetchContent_Declare(mbedtls
    GIT_REPOSITORY https://github.com/Mbed-TLS/mbedtls.git
    GIT_TAG 0bebf8b8c7f07abe3571ded48a11aa907a1ffb20
    GIT_SHALLOW FALSE
    GIT_SUBMODULES_RECURSE TRUE)
  FetchContent_MakeAvailable(mbedtls)
endif()
add_library(nb_core src/core.c src/history.c src/html.c src/layout.c src/http.c)
target_include_directories(nb_core PUBLIC src)
if(WIN32)
  target_sources(nb_core PRIVATE src/network.c src/tls.c)
  target_link_libraries(nb_core PUBLIC winhttp ws2_32 crypt32 mbedtls mbedx509 mbedcrypto)
  add_executable(NetBeyond WIN32 src/netbeyond.c)
  target_link_libraries(NetBeyond PRIVATE nb_core)
endif()
foreach(test_name core history html layout http)
  add_executable(test_${test_name} tests/test_${test_name}.c)
  target_link_libraries(test_${test_name} PRIVATE nb_core)
endforeach()
enable_testing()
foreach(test_name core history html layout http)
  add_test(NAME ${test_name} COMMAND test_${test_name})
endforeach()
