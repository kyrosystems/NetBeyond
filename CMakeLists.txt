cmake_minimum_required(VERSION 3.16)
project(NetBeyond C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(nb_core src/core.c src/history.c src/html.c src/layout.c)
target_include_directories(nb_core PUBLIC src)
if(WIN32)
 add_executable(NetBeyond WIN32 src/netbeyond.c)
 target_link_libraries(NetBeyond PRIVATE nb_core ole32 oleaut32 uuid)
endif()
foreach(t core history html layout)
 add_executable(test_${t} tests/test_${t}.c)
 target_link_libraries(test_${t} PRIVATE nb_core)
endforeach()
enable_testing()
foreach(t core history html layout)
 add_test(NAME ${t} COMMAND test_${t})
endforeach()
